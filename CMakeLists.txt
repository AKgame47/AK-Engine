cmake_minimum_required(VERSION 3.16)
project(AKEngine VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(ak_core core/src/core.cpp)
target_include_directories(ak_core PUBLIC core/include)
add_executable(ak_core_tests core/tests/tests.cpp)
target_link_libraries(ak_core_tests ak_core)
add_test(NAME ak_core_tests COMMAND ak_core_tests)
